# Priority #10 — canonical centralized/local/transfer baselines

from dataclasses import replace
from pathlib import Path
import pandas as pd

from ecg_revision.common import Config, code_hash, read_json
from ecg_revision.data import Store
from ecg_revision.training import run
from ecg_revision.evaluation import evaluate_d2

base_dir = Path(BASE_DIR)
root = base_dir / "study"
fast = "/content/ecg_fl_revision_fast_cache"

base = Config(
    rounds=20,
    batch_size=BATCH_SIZE,
    architecture="manuscript_bn",
    schedule="sampled_epoch",
    imbalance="sampler_only",
    selection="macro_domain_auc",
    split_policy="patient42",
    unknown_policy="exclude",
    train_tau=0.05,
)
base.validate()

store = Store(
    base_dir / "data",
    legacy_root=LEGACY_ROOT,
    split_policy="patient42",
    unknown_policy="exclude",
)

# provenance guard
selected = read_json(root / "selected_hyperparameters.json")
assert selected["data_id"] == store.data_id
assert selected["code_hash"] == code_hash()

# Non-FL protocol: 5 epochs
b = replace(
    base,
    rounds=5,
    local_epochs=1,
    weighting="equal",
    phase="report",
)

runs = []

print("Priority #10 canonical baselines")
print("Centralized pooled: seeds 0-9, 5 epochs")
print("PTB-only / INCART-only / transfer: seeds 0-2, 5 epochs per stage")
print("Checkpoint selection is validation-only.")

# 1) pooled centralized — 10 seeds
for seed in range(10):
    cfg = replace(
        b,
        method="pooled",
        seed=seed,
        selection="macro_domain_auc",
    )
    path = run(cfg, store, root, fast)
    done = read_json(path / "complete.json")
    runs.append({
        "method": "pooled",
        "seed": seed,
        "run_path": str(path),
        "best_round": done["best_round"],
        "best_validation": done["best_score"],
    })

# 2) local-only + transfer — matched seeds 0,1,2
for seed in (0, 1, 2):

    ptb_cfg = replace(
        b,
        method="ptb_only",
        seed=seed,
        selection="ptb_auc",
    )
    ptb_path = run(ptb_cfg, store, root, fast)
    done = read_json(ptb_path / "complete.json")
    runs.append({
        "method": "ptb_only",
        "seed": seed,
        "run_path": str(ptb_path),
        "best_round": done["best_round"],
        "best_validation": done["best_score"],
    })

    inc_cfg = replace(
        b,
        method="incart_only",
        seed=seed,
        selection="incart_auc",
    )
    inc_path = run(inc_cfg, store, root, fast)
    done = read_json(inc_path / "complete.json")
    runs.append({
        "method": "incart_only",
        "seed": seed,
        "run_path": str(inc_path),
        "best_round": done["best_round"],
        "best_validation": done["best_score"],
    })

    transfer_cfg = replace(
        b,
        method="transfer",
        seed=seed,
        selection="incart_auc",
    )
    transfer_path = run(
        transfer_cfg,
        store,
        root,
        fast,
        initial_checkpoint=ptb_path / "best.pt",
    )
    done = read_json(transfer_path / "complete.json")
    runs.append({
        "method": "transfer",
        "seed": seed,
        "run_path": str(transfer_path),
        "best_round": done["best_round"],
        "best_validation": done["best_score"],
    })

runs = pd.DataFrame(runs)
runs.to_csv(root / "baseline_canonical_runs.csv", index=False)

# Held-out evaluation
rows = []

for r in runs.itertuples(index=False):
    df = evaluate_d2(
        Path(r.run_path) / "best.pt",
        store,
        root,
        bootstraps=BOOTSTRAPS,
        fast_root=fast,
    )

    # canonical INCART label definition tau=0.05
    df = df[
        (df.domain == "PTB") |
        ((df.domain == "INCART") & (df.eval_tau == 0.05))
    ]

    rows.append(df)

results = pd.concat(rows, ignore_index=True)

metrics = [
    "roc_auc",
    "average_precision",
    "mcc",
    "f1",
    "balanced_accuracy",
    "sensitivity",
    "ppv",
]

summary = (
    results
    .groupby(["method", "domain"])
    .agg(
        n_seeds=("seed", "nunique"),
        roc_auc_mean=("roc_auc", "mean"),
        roc_auc_sd=("roc_auc", "std"),
        pr_auc_mean=("average_precision", "mean"),
        pr_auc_sd=("average_precision", "std"),
        mcc_mean=("mcc", "mean"),
        mcc_sd=("mcc", "std"),
        f1_mean=("f1", "mean"),
        f1_sd=("f1", "std"),
    )
    .reset_index()
)

results.to_csv(root / "baseline_canonical_evaluation_rows.csv", index=False)
summary.to_csv(root / "baseline_canonical_summary.csv", index=False)

print("\nBASELINE CANONICAL SUMMARY")
print(summary.to_string(index=False))
print("\nNo held-out test data were used for training or checkpoint selection.")
